# only 1st time
git clone https://github.com/Dede20231/LATAM.git
cd LATAM

# every time
mkdir DEST_REPO (create the Repository on Github)
cd DEST_REPO 
git init

mv ../*.* .
git status
git add .
git commit -m "Initial commit - imported source LATAM"
git remote add origin https://github.com/Dede20231/DEST_REPO.git
git remote -v

unset GITHUB_TOKEN
gh auth login
git branch -M main
git push -u origin main

# https://github.com/settings/tokens ( get the token key and set it as password )

⚠️ Common issues
❌ “refusing to merge unrelated histories”

Fix:

git pull origin main --allow-unrelated-histories
❌ remote already exists
git remote remove origin
